cmake_minimum_required(VERSION 3.16)
project(youspeed_storage_prototype LANGUAGES C CXX)
option(YSP_WITH_GEODESK "Compile GeoDesk native comparator" OFF)
find_package(SQLite3 REQUIRED)
find_package(ZLIB REQUIRED)
add_library(ysp_zstd_decoder STATIC zstd_decoder.c)
set_target_properties(ysp_zstd_decoder PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(packed_runtime STATIC packed_runtime.cpp)
target_compile_features(packed_runtime PUBLIC cxx_std_17)
target_include_directories(packed_runtime PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(packed_runtime PUBLIC ZLIB::ZLIB ysp_zstd_decoder)
add_library(packed_runtime_test_api SHARED packed_runtime.cpp)
target_compile_features(packed_runtime_test_api PUBLIC cxx_std_17)
target_link_libraries(packed_runtime_test_api PRIVATE ZLIB::ZLIB ysp_zstd_decoder)
add_library(packed_runtime_vfs STATIC packed_sqlite_vfs.cpp)
target_compile_features(packed_runtime_vfs PUBLIC cxx_std_17)
target_link_libraries(packed_runtime_vfs PUBLIC packed_runtime SQLite::SQLite3)
add_executable(packed-runtime-check packed_runtime_check.cpp)
target_link_libraries(packed-runtime-check PRIVATE packed_runtime_vfs)
add_library(runtime_storage STATIC runtime_storage.cpp)
target_compile_features(runtime_storage PUBLIC cxx_std_17)
target_include_directories(runtime_storage PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(runtime_storage PUBLIC SQLite::SQLite3 ZLIB::ZLIB)
if(YSP_WITH_GEODESK)
  add_subdirectory(../runtime-storage-geodesk geodesk)
  target_compile_definitions(runtime_storage PRIVATE YSP_WITH_GEODESK=1)
  target_link_libraries(runtime_storage PUBLIC storage_geodesk)
  target_include_directories(runtime_storage PRIVATE ../runtime-storage-geodesk)
endif()
add_executable(rsbench main.cpp)
target_link_libraries(rsbench PRIVATE runtime_storage)
