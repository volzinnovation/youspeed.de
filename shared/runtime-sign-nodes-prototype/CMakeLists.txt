cmake_minimum_required(VERSION 3.16)
project(youspeed_sign_nodes_prototype LANGUAGES C CXX)
find_package(SQLite3 REQUIRED)
find_package(ZLIB REQUIRED)
add_library(packed_nodes_test_api SHARED runtime_sign_nodes.cpp)
target_compile_features(packed_nodes_test_api PRIVATE cxx_std_17)
target_compile_definitions(packed_nodes_test_api PRIVATE YSP_NODES_PACKED_ONLY=1)
target_link_libraries(packed_nodes_test_api PRIVATE ZLIB::ZLIB)
add_library(runtime_sign_nodes STATIC runtime_sign_nodes.cpp)
target_compile_features(runtime_sign_nodes PUBLIC cxx_std_17)
target_include_directories(runtime_sign_nodes PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(runtime_sign_nodes PUBLIC SQLite::SQLite3 ZLIB::ZLIB)
add_executable(nodebench main.cpp)
target_link_libraries(nodebench PRIVATE runtime_sign_nodes)
