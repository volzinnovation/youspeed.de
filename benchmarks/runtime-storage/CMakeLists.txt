cmake_minimum_required(VERSION 3.20)
project(youspeed_phone_storage_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SQLITE_AMALGAMATION_DIR "" CACHE PATH "Official pinned SQLite 3.51.0 source")
if(NOT EXISTS "${SQLITE_AMALGAMATION_DIR}/sqlite3.c")
  message(FATAL_ERROR "Set SQLITE_AMALGAMATION_DIR to the verified SQLite amalgamation")
endif()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_library(benchmark_sqlite STATIC "${SQLITE_AMALGAMATION_DIR}/sqlite3.c")
target_include_directories(benchmark_sqlite PUBLIC "${SQLITE_AMALGAMATION_DIR}")
target_compile_definitions(benchmark_sqlite PRIVATE SQLITE_ENABLE_RTREE=1 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_THREADSAFE=1)
find_package(ZLIB REQUIRED)
add_subdirectory("${REPO_ROOT}/shared/runtime-storage-geodesk" geodesk)
add_library(benchmark_storage STATIC "${REPO_ROOT}/shared/runtime-storage-prototype/runtime_storage.cpp")
target_include_directories(benchmark_storage PUBLIC "${REPO_ROOT}/shared/runtime-storage-prototype")
target_include_directories(benchmark_storage PRIVATE "${REPO_ROOT}/shared/runtime-storage-geodesk")
target_compile_definitions(benchmark_storage PRIVATE YSP_WITH_GEODESK=1)
target_link_libraries(benchmark_storage PUBLIC benchmark_sqlite ZLIB::ZLIB storage_geodesk)
add_library(benchmark_corrections STATIC "${REPO_ROOT}/shared/runtime-corrections-prototype/runtime_corrections.cpp")
# The generic-node module is shared by the standalone Android runner and iOS app.
file(GLOB NODE_SOURCES "${REPO_ROOT}/shared/runtime-sign-nodes-prototype/*.cpp")
list(FILTER NODE_SOURCES EXCLUDE REGEX "(test|main)\\.cpp$")
add_library(benchmark_nodes STATIC ${NODE_SOURCES})
target_compile_definitions(benchmark_nodes PRIVATE YSP_WITH_GEODESK=1)
target_include_directories(benchmark_nodes PRIVATE "${REPO_ROOT}/shared/runtime-storage-geodesk" "${SQLITE_AMALGAMATION_DIR}")
target_link_libraries(benchmark_nodes PUBLIC benchmark_sqlite ZLIB::ZLIB storage_geodesk)
add_executable(storage_phone_runner phone_main.cpp)
target_link_libraries(storage_phone_runner PRIVATE benchmark_storage benchmark_nodes benchmark_corrections)
